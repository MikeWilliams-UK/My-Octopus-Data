-- Useful SQLite queries

SELECT StartTime,
       EndTime,
       EnergyAdded,
       TypeOfCharge,
       Problems
FROM ChargeEvents
WHERE EnergyAdded > 0
  AND StartTime > '2026-05-26'
ORDER BY StartTime DESC

-- Charge Events
SELECT MIN(StartTime) [First Event],
       MAX(EndTime) [Last Event],
       COUNT(EnergyAdded) Events,
       SUM(EnergyAdded) [kWh Added]
FROM ChargeEvents
WHERE EnergyAdded > 0
AND StartTime BETWEEN '2026-08-03' AND '2026-09-30'
GROUP BY SUBSTR(StartTime, 1, 11)
ORDER BY StartTime DESC

-- EV Charge Costs
Select *
FROM HalfHourlyCostsElectric
Where 1=1
AND Consumption > 0
AND SUBSTR(CostType, 1, 2) = 'EV'
AND StartTime BETWEEN '2026-08-16 10' AND '2026-08-16 16'
ORDER BY StartTime DESC

-- All Electric Costs for a free hour
Select *
FROM HalfHourlyCostsElectric
Where 1=1
AND Consumption > 0
AND StartTime > '2026-09-06 13' AND StartTime < '2026-09-06 15'
ORDER BY StartTime, CostType

-- All Gas Charges
Select *
FROM HalfHourlyCostsGas
Where 1=1
AND Consumption > 0
AND StartTime > '2026-09-01'
ORDER BY StartTime DESC
